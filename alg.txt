import math
import time
import pygame
from .connection import Connection
from .drone import Drone
from .graph import Graph
from .hub import Hub


class SimulationGUI:
    """Renders the 2D visual simulation using Pygame."""

    # Color palette mapping
    COLOR_MAP: dict[str, tuple[int, int, int]] = {
        "white": (255, 255, 255),
        "black": (0, 0, 0),
        "red": (220, 53, 69),
        "green": (40, 167, 69),
        "blue": (0, 123, 255),
        "yellow": (255, 193, 7),
        "gray": (108, 117, 125),
        "cyan": (23, 162, 184),
        "orange": (253, 126, 20),
    }

    def __init__(
        self,
        graph: Graph,
        width: int = 1000,
        height: int = 800,
    ) -> None:
        """Initialize Pygame display and calculate coordinate scaling."""
        pygame.init()
        pygame.font.init()

        self._graph: Graph = graph
        self._width: int = width
        self._height: int = height
        self._screen: pygame.Surface = pygame.display.set_mode((width, height))
        pygame.display.set_caption("Fly-in Simulation")

        self._font: pygame.font.Font = pygame.font.SysFont("Arial", 18)
        self._title_font: pygame.font.Font = pygame.font.SysFont("Arial", 22, bold=True)

        self._start_time: float = time.time()
        self._current_turn: int = 0
        self._drones: list[Drone] = []

        # Pre-compute layout scaling to map Cartesian coordinates to window pixels
        self._padding: int = 100
        self._min_x, self._max_x, self._min_y, self._max_y = self._get_coordinate_bounds()

    def _get_coordinate_bounds(self) -> tuple[int, int, int, int]:
        """Find graph bounds to scale Cartesian points onto the window."""
        x_coords = [hub.coordinates[0] for hub in self._graph._hubs.values()]
        y_coords = [hub.coordinates[1] for hub in self._graph._hubs.values()]

        if not x_coords or not y_coords:
            return 0, 1, 0, 1

        min_x, max_x = min(x_coords), max(x_coords)
        min_y, max_y = min(y_coords), max(y_coords)

        # Avoid division by zero for single-point graphs or straight lines
        if min_x == max_x:
            max_x += 1
        if min_y == max_y:
            max_y += 1

        return min_x, max_x, min_y, max_y

    def _map_to_screen(self, x: float, y: float) -> tuple[int, int]:
        """Convert graph Cartesian coordinates to screen pixel coordinates."""
        draw_width = self._width - 2 * self._padding
        draw_height = self._height - 2 * self._padding

        norm_x = (x - self._min_x) / (self._max_x - self._min_x)
        norm_y = (y - self._min_y) / (self._max_y - self._min_y)

        screen_x = int(self._padding + norm_x * draw_width)
        # Invert Y axis so higher Y coordinates go up on the screen
        screen_y = int(self._height - (self._padding + norm_y * draw_height))

        return screen_x, screen_y

    def update_simulation_state(self, turn: int, drones: list[Drone]) -> None:
        """Update current turn and drone list for frame rendering."""
        self._current_turn = turn
        self._drones = drones

    def _get_rgb_color(self, color_name: str, default: tuple[int, int, int]) -> tuple[int, int, int]:
        """Convert color name string to an RGB tuple."""
        return self.COLOR_MAP.get(color_name.lower(), default)

    def _draw_dashed_line(
        self,
        color: tuple[int, int, int],
        start: tuple[int, int],
        end: tuple[int, int],
        dash_len: int = 10,
        space_len: int = 5,
    ) -> None:
        """Draw a dashed line between two points."""
        dx = end[0] - start[0]
        dy = end[1] - start[1]
        distance = math.hypot(dx, dy)
        if distance == 0:
            return

        angle = math.atan2(dy, dx)
        step = dash_len + space_len
        num_dashes = int(distance // step)

        for i in range(num_dashes + 1):
            start_dist = i * step
            end_dist = min(start_dist + dash_len, distance)

            p1 = (
                int(start[0] + math.cos(angle) * start_dist),
                int(start[1] + math.sin(angle) * start_dist),
            )
            p2 = (
                int(start[0] + math.cos(angle) * end_dist),
                int(start[1] + math.sin(angle) * end_dist),
            )
            pygame.draw.line(self._screen, color, p1, p2, 2)

    def _draw_dotted_line(
        self,
        color: tuple[int, int, int],
        start: tuple[int, int],
        end: tuple[int, int],
        dot_spacing: int = 8,
    ) -> None:
        """Draw a dotted line between two points."""
        dx = end[0] - start[0]
        dy = end[1] - start[1]
        distance = math.hypot(dx, dy)
        if distance == 0:
            return

        angle = math.atan2(dy, dx)
        num_dots = int(distance // dot_spacing)

        for i in range(num_dots + 1):
            dist = i * dot_spacing
            p = (
                int(start[0] + math.cos(angle) * dist),
                int(start[1] + math.sin(angle) * dist),
            )
            pygame.draw.circle(self._screen, color, p, 2)

    def _draw_cross_line(
        self,
        color: tuple[int, int, int],
        start: tuple[int, int],
        end: tuple[int, int],
        spacing: int = 20,
    ) -> None:
        """Draw a line with cross marks for blocked connections."""
        pygame.draw.line(self._screen, color, start, end, 1)

        dx = end[0] - start[0]
        dy = end[1] - start[1]
        distance = math.hypot(dx, dy)
        if distance == 0:
            return

        angle = math.atan2(dy, dx)
        num_crosses = int(distance // spacing)

        for i in range(1, num_crosses + 1):
            dist = i * spacing
            cx = start[0] + math.cos(angle) * dist
            cy = start[1] + math.sin(angle) * dist

            cross_size = 4
            p1 = (int(cx - cross_size), int(cy - cross_size))
            p2 = (int(cx + cross_size), int(cy + cross_size))
            p3 = (int(cx - cross_size), int(cy + cross_size))
            p4 = (int(cx + cross_size), int(cy - cross_size))

            pygame.draw.line(self._screen, color, p1, p2, 2)
            pygame.draw.line(self._screen, color, p3, p4, 2)

    def _draw_connections(self) -> None:
        """Render all connections between hubs (Layer 1 - Bottom)."""
        for conn in self._graph._connections:
            pos1 = self._map_to_screen(*conn.hub1.coordinates)
            pos2 = self._map_to_screen(*conn.hub2.coordinates)

            # Style connection based on target hub zone type
            target_zone = conn.hub2.zone

            if target_zone == "priority":
                # Dotted green line
                self._draw_dotted_line(self.COLOR_MAP["green"], pos1, pos2)
            elif target_zone == "restricted":
                # Dashed yellow line
                self._draw_dashed_line(self.COLOR_MAP["yellow"], pos1, pos2)
            elif target_zone == "blocked":
                # Red line with crosses
                self._draw_cross_line(self.COLOR_MAP["red"], pos1, pos2)
            else:
                # Solid white line (Normal)
                pygame.draw.line(self._screen, self.COLOR_MAP["white"], pos1, pos2, 2)

    def _draw_hubs(self) -> None:
        """Render all hubs as colored circles (Layer 2 - Middle)."""
        for hub in self._graph._hubs.values():
            pos = self._map_to_screen(*hub.coordinates)
            hub_color = self._get_rgb_color(hub._color, self.COLOR_MAP["blue"])

            # Outer boundary and filled circle
            pygame.draw.circle(self._screen, hub_color, pos, 20)
            pygame.draw.circle(self._screen, self.COLOR_MAP["white"], pos, 20, 2)

            # Hub label text
            text_surf = self._font.render(hub.name, True, self.COLOR_MAP["white"])
            text_rect = text_surf.get_rect(center=(pos[0], pos[1] - 30))
            self._screen.blit(text_surf, text_rect)

    def _draw_triangle_drone(self, pos: tuple[int, int], size: int = 12) -> None:
        """Draw an isosceles triangle representing a drone."""
        x, y = pos
        points = [
            (x, y - size),  # Top vertex
            (x - size, y + size),  # Bottom left
            (x + size, y + size),  # Bottom right
        ]
        pygame.draw.polygon(self._screen, self.COLOR_MAP["cyan"], points)
        pygame.draw.polygon(self._screen, self.COLOR_MAP["black"], points, 1)

    def _draw_drones(self) -> None:
        """Render drones positioned at their current hubs (Layer 3 - Top)."""
        for drone in self._drones:
            if not drone.is_active:
                continue

            # Position drone at its current hub
            current_hub = drone.current_hub
            pos = self._map_to_screen(*current_hub.coordinates)

            # Draw drone triangle overlapping hubs and links
            self._draw_triangle_drone(pos)

            # Render drone identifier text
            label = self._font.render(drone.name, True, self.COLOR_MAP["black"])
            self._screen.blit(label, (pos[0] - 8, pos[1] - 5))

    def _draw_hud_info(self) -> None:
        """Render HUD displaying elapsed time and turn counter (Layer 4 - HUD)."""
        elapsed_sec = time.time() - self._start_time

        time_str = f"Elapsed Time: {elapsed_sec:.1f}s"
        turn_str = f"Turns: {self._current_turn}"

        time_surf = self._title_font.render(time_str, True, self.COLOR_MAP["white"])
        turn_surf = self._title_font.render(turn_str, True, self.COLOR_MAP["white"])

        self._screen.blit(time_surf, (20, 20))
        self._screen.blit(turn_surf, (20, 50))

    def render_frame(self) -> bool:
        """Process Pygame events and render one frame. Returns False on quit."""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                pygame.quit()
                return False

        # Clear background with dark grey
        self._screen.fill((30, 30, 30))

        # Render layers in priority sequence
        self._draw_connections()
        self._draw_hubs()
        self._draw_drones()
        self._draw_hud_info()

        pygame.display.flip()
        return True
